function preparedConstructorFields(ctx: EvaluateContext, value: unknown, names: string[]): unknown[] {
  const result = invokeCallable({ $ref: "std:representationSnapshotConstructorFields" }, [representationValueSnapshot(value), names], ctx) as { $tag?: string; _0?: unknown };
  if (result?.$tag !== "Just" || !Array.isArray(result._0)) throw new TypeError("Invalid constructor payload at the representation boundary");
  return result._0;
}

function integerMarkerNative(ctx: EvaluateContext, marker: unknown, owner: "Nat" | "Int"): unknown {
  const [carrier] = preparedConstructorFields(ctx, marker, ["carrier"]);
  const tag = Object.getOwnPropertyDescriptor(carrier as object, "$tag")?.value;
  if (tag === "IntegerSmall") return preparedConstructorFields(ctx, carrier, ["value"])[0];
  if (tag === "IntegerDigits") return { $tag: owner, digits: preparedConstructorFields(ctx, carrier, ["digits"])[0] };
  throw new TypeError("Invalid integer carrier constructor at the representation boundary");
}


export const originalMarkerLeaves = {
  integerNatNativeFromMarker: (ctx, marker) => integerMarkerNative(ctx, marker, "Nat"),
  integerIntNativeFromMarker: (ctx, marker) => integerMarkerNative(ctx, marker, "Int"),
  rationalNativeFromMarker: (ctx, marker) => {
    const [num, den] = preparedConstructorFields(ctx, marker, ["num", "den"]);
    return { $tag: "Rational", num, den };
  },
  typeUnionFromMarker: (ctx, marker): FenelType => {
    const [members] = preparedConstructorFields(ctx, marker, ["members"]);
    return { $union: members as FenelType[] };
  },
  taggedSingletonTypeFromMarker: (ctx, marker): FenelType => {
    const [base, phantomBase, constraint] = preparedConstructorFields(ctx, marker, ["base", "phantomBase", "constraint"]);
    return { $tagged: { value: base as FenelType, phantom: { $refined: { base: phantomBase as FenelType, set: constraint as SetExpr } } } };
  },

};
